"""
Joker Poker Board Game - Full Implementation

5-card board-only poker with up to 2 wild Jokers.
13 hand types from High Card to Royal Flush.
"""

import random
from typing import List, Dict, Any, Optional, Tuple
from dataclasses import dataclass
from enum import Enum
from collections import Counter

# ============================================================================
# CARD TYPES
# ============================================================================

class Rank(Enum):
    TWO = "2"
    THREE = "3"
    FOUR = "4"
    FIVE = "5"
    SIX = "6"
    SEVEN = "7"
    EIGHT = "8"
    NINE = "9"
    TEN = "10"
    JACK = "J"
    QUEEN = "Q"
    KING = "K"
    ACE = "A"
    JOKER = "JOKER"

class Suit(Enum):
    CLUBS = "C"
    DIAMONDS = "D"
    HEARTS = "H"
    SPADES = "S"

@dataclass(frozen=True)
class Card:
    rank: Rank
    suit: Optional[Suit]  # None for Joker
    
    def __str__(self) -> str:
        if self.rank == Rank.JOKER:
            return "JOKER"
        return f"{self.rank.value}{self.suit.value}"
    
    @property
    def is_joker(self) -> bool:
        return self.rank == Rank.JOKER
    
    @property
    def numeric_value(self) -> int:
        """For sorting and comparison"""
        values = {
            Rank.TWO: 2, Rank.THREE: 3, Rank.FOUR: 4, Rank.FIVE: 5,
            Rank.SIX: 6, Rank.SEVEN: 7, Rank.EIGHT: 8, Rank.NINE: 9,
            Rank.TEN: 10, Rank.JACK: 11, Rank.QUEEN: 12, Rank.KING: 13,
            Rank.ACE: 14, Rank.JOKER: 0
        }
        return values[self.rank]

# Standard ranks (no Joker)
STANDARD_RANKS = [r for r in Rank if r != Rank.JOKER]
SUITS = list(Suit)

def create_deck() -> List[Card]:
    """Create 54-card deck: 52 standard + 2 Jokers"""
    deck = []
    
    # 52 standard cards
    for suit in SUITS:
        for rank in STANDARD_RANKS:
            deck.append(Card(rank=rank, suit=suit))
    
    # 2 Jokers
    deck.append(Card(rank=Rank.JOKER, suit=None))
    deck.append(Card(rank=Rank.JOKER, suit=None))
    
    assert len(deck) == 54
    return deck

def get_all_standard_cards() -> List[Card]:
    """Get all 52 standard cards (no Jokers)"""
    cards = []
    for suit in SUITS:
        for rank in STANDARD_RANKS:
            cards.append(Card(rank=rank, suit=suit))
    return cards

# ============================================================================
# HAND EVALUATION
# ============================================================================

class HandType(Enum):
    ROYAL_FLUSH = "ROYAL_FLUSH"
    STRAIGHT_FLUSH = "STRAIGHT_FLUSH"
    FOUR_OF_A_KIND = "FOUR_OF_A_KIND"
    FULL_HOUSE = "FULL_HOUSE"
    FLUSH = "FLUSH"
    STRAIGHT = "STRAIGHT"
    THREE_OF_A_KIND = "THREE_OF_A_KIND"
    TWO_PAIR = "TWO_PAIR"
    PAIR = "PAIR"
    HIGH_CARD = "HIGH_CARD"

class RankCategory(Enum):
    LOW = "LOW"    # 2-10
    HIGH = "HIGH"  # J-A
    ANY = "ANY"    # Doesn't matter

@dataclass
class HandResult:
    hand_type: HandType
    rank_category: RankCategory
    primary_rank: Rank
    kickers: List[Rank]
    winning_positions: List[int]
    multiplier: float

# Paytable (multipliers) - Optimized for 97% RTP
PAYTABLE = {
    HandType.ROYAL_FLUSH: {RankCategory.ANY: 1000.0},
    HandType.STRAIGHT_FLUSH: {RankCategory.ANY: 100.0},
    HandType.FOUR_OF_A_KIND: {RankCategory.ANY: 40.0},
    HandType.FULL_HOUSE: {RankCategory.ANY: 20.0},
    HandType.FLUSH: {RankCategory.ANY: 10.0},
    HandType.STRAIGHT: {RankCategory.ANY: 5.0},
    HandType.THREE_OF_A_KIND: {
        RankCategory.LOW: 1.8,
        RankCategory.HIGH: 3.0,
    },
    HandType.TWO_PAIR: {
        RankCategory.LOW: 0.8,
        RankCategory.HIGH: 1.5,
    },
    HandType.PAIR: {
        RankCategory.LOW: 0.55,
        RankCategory.HIGH: 0.9,
    },
    HandType.HIGH_CARD: {
        RankCategory.LOW: 0.1,
        RankCategory.HIGH: 0.35,
    },
}

def get_rank_category(rank: Rank) -> RankCategory:
    """Determine if rank is LOW (2-10) or HIGH (J-A)"""
    face_cards = {Rank.JACK, Rank.QUEEN, Rank.KING, Rank.ACE}
    return RankCategory.HIGH if rank in face_cards else RankCategory.LOW

def evaluate_hand(cards: List[Card]) -> HandResult:
    """
    Evaluate 5 cards and return best poker hand.
    
    Cards must NOT contain Jokers - resolve them first!
    """
    if len(cards) != 5:
        raise ValueError(f"Must have 5 cards, got {len(cards)}")
    
    if any(c.is_joker for c in cards):
        raise ValueError("Jokers must be resolved before evaluation")
    
    # Check hands from strongest to weakest
    if result := _check_royal_flush(cards):
        return result
    if result := _check_straight_flush(cards):
        return result
    if result := _check_four_of_a_kind(cards):
        return result
    if result := _check_full_house(cards):
        return result
    if result := _check_flush(cards):
        return result
    if result := _check_straight(cards):
        return result
    if result := _check_three_of_a_kind(cards):
        return result
    if result := _check_two_pair(cards):
        return result
    if result := _check_pair(cards):
        return result
    
    return _check_high_card(cards)

def _check_royal_flush(cards: List[Card]) -> Optional[HandResult]:
    """10-J-Q-K-A of same suit"""
    if not _is_flush(cards):
        return None
    
    royal_ranks = {Rank.TEN, Rank.JACK, Rank.QUEEN, Rank.KING, Rank.ACE}
    card_ranks = {c.rank for c in cards}
    
    if card_ranks == royal_ranks:
        return HandResult(
            hand_type=HandType.ROYAL_FLUSH,
            rank_category=RankCategory.ANY,
            primary_rank=Rank.ACE,
            kickers=[],
            winning_positions=[0, 1, 2, 3, 4],
            multiplier=PAYTABLE[HandType.ROYAL_FLUSH][RankCategory.ANY]
        )
    return None

def _check_straight_flush(cards: List[Card]) -> Optional[HandResult]:
    """Straight + Flush (not Royal)"""
    if not _is_flush(cards):
        return None
    
    if high_card := _is_straight(cards):
        return HandResult(
            hand_type=HandType.STRAIGHT_FLUSH,
            rank_category=RankCategory.ANY,
            primary_rank=high_card,
            kickers=[],
            winning_positions=[0, 1, 2, 3, 4],
            multiplier=PAYTABLE[HandType.STRAIGHT_FLUSH][RankCategory.ANY]
        )
    return None

def _check_four_of_a_kind(cards: List[Card]) -> Optional[HandResult]:
    """4 cards of same rank"""
    rank_counts = Counter(c.rank for c in cards)
    
    for rank, count in rank_counts.items():
        if count == 4:
            positions = [i for i, c in enumerate(cards) if c.rank == rank]
            kicker = [r for r, cnt in rank_counts.items() if cnt == 1][0]
            
            return HandResult(
                hand_type=HandType.FOUR_OF_A_KIND,
                rank_category=RankCategory.ANY,
                primary_rank=rank,
                kickers=[kicker],
                winning_positions=positions,
                multiplier=PAYTABLE[HandType.FOUR_OF_A_KIND][RankCategory.ANY]
            )
    return None

def _check_full_house(cards: List[Card]) -> Optional[HandResult]:
    """3 of a kind + pair"""
    rank_counts = Counter(c.rank for c in cards)
    
    three_rank = None
    pair_rank = None
    
    for rank, count in rank_counts.items():
        if count == 3:
            three_rank = rank
        elif count == 2:
            pair_rank = rank
    
    if three_rank and pair_rank:
        positions = [i for i, c in enumerate(cards) if c.rank in (three_rank, pair_rank)]
        
        return HandResult(
            hand_type=HandType.FULL_HOUSE,
            rank_category=RankCategory.ANY,
            primary_rank=three_rank,
            kickers=[pair_rank],
            winning_positions=positions,
            multiplier=PAYTABLE[HandType.FULL_HOUSE][RankCategory.ANY]
        )
    return None

def _check_flush(cards: List[Card]) -> Optional[HandResult]:
    """All same suit"""
    if _is_flush(cards):
        sorted_cards = sorted(cards, key=lambda c: c.numeric_value, reverse=True)
        high_card = sorted_cards[0].rank
        
        return HandResult(
            hand_type=HandType.FLUSH,
            rank_category=RankCategory.ANY,
            primary_rank=high_card,
            kickers=[c.rank for c in sorted_cards[1:]],
            winning_positions=[0, 1, 2, 3, 4],
            multiplier=PAYTABLE[HandType.FLUSH][RankCategory.ANY]
        )
    return None

def _check_straight(cards: List[Card]) -> Optional[HandResult]:
    """5 consecutive ranks"""
    if high_card := _is_straight(cards):
        return HandResult(
            hand_type=HandType.STRAIGHT,
            rank_category=RankCategory.ANY,
            primary_rank=high_card,
            kickers=[],
            winning_positions=[0, 1, 2, 3, 4],
            multiplier=PAYTABLE[HandType.STRAIGHT][RankCategory.ANY]
        )
    return None

def _check_three_of_a_kind(cards: List[Card]) -> Optional[HandResult]:
    """3 cards of same rank"""
    rank_counts = Counter(c.rank for c in cards)
    
    for rank, count in rank_counts.items():
        if count == 3:
            positions = [i for i, c in enumerate(cards) if c.rank == rank]
            kickers = sorted([r for r, cnt in rank_counts.items() if cnt == 1],
                           key=lambda r: r.value, reverse=True)
            
            category = get_rank_category(rank)
            multiplier = PAYTABLE[HandType.THREE_OF_A_KIND][category]
            
            return HandResult(
                hand_type=HandType.THREE_OF_A_KIND,
                rank_category=category,
                primary_rank=rank,
                kickers=kickers,
                winning_positions=positions,
                multiplier=multiplier
            )
    return None

def _check_two_pair(cards: List[Card]) -> Optional[HandResult]:
    """2 pairs"""
    rank_counts = Counter(c.rank for c in cards)
    pairs = [rank for rank, count in rank_counts.items() if count == 2]
    
    if len(pairs) == 2:
        pairs_sorted = sorted(pairs, key=lambda r: r.value, reverse=True)
        high_pair = pairs_sorted[0]
        positions = [i for i, c in enumerate(cards) if c.rank in pairs]
        kicker = [r for r, cnt in rank_counts.items() if cnt == 1][0]
        
        category = get_rank_category(high_pair)
        multiplier = PAYTABLE[HandType.TWO_PAIR][category]
        
        return HandResult(
            hand_type=HandType.TWO_PAIR,
            rank_category=category,
            primary_rank=high_pair,
            kickers=[pairs_sorted[1], kicker],
            winning_positions=positions,
            multiplier=multiplier
        )
    return None

def _check_pair(cards: List[Card]) -> Optional[HandResult]:
    """One pair"""
    rank_counts = Counter(c.rank for c in cards)
    
    for rank, count in rank_counts.items():
        if count == 2:
            positions = [i for i, c in enumerate(cards) if c.rank == rank]
            kickers = sorted([r for r, cnt in rank_counts.items() if cnt == 1],
                           key=lambda r: r.value, reverse=True)
            
            category = get_rank_category(rank)
            multiplier = PAYTABLE[HandType.PAIR][category]
            
            return HandResult(
                hand_type=HandType.PAIR,
                rank_category=category,
                primary_rank=rank,
                kickers=kickers,
                winning_positions=positions,
                multiplier=multiplier
            )
    return None

def _check_high_card(cards: List[Card]) -> HandResult:
    """No other hand - high card only"""
    sorted_cards = sorted(cards, key=lambda c: c.numeric_value, reverse=True)
    high_card = sorted_cards[0]
    
    category = get_rank_category(high_card.rank)
    multiplier = PAYTABLE[HandType.HIGH_CARD][category]
    
    positions = [0] if category == RankCategory.HIGH else []
    
    return HandResult(
        hand_type=HandType.HIGH_CARD,
        rank_category=category,
        primary_rank=high_card.rank,
        kickers=[c.rank for c in sorted_cards[1:]],
        winning_positions=positions,
        multiplier=multiplier
    )

def _is_flush(cards: List[Card]) -> bool:
    """Check if all cards are same suit"""
    suits = [c.suit for c in cards]
    return len(set(suits)) == 1

def _is_straight(cards: List[Card]) -> Optional[Rank]:
    """Check if cards form straight. Returns high card if yes, None if no."""
    values = sorted([c.numeric_value for c in cards])
    
    # Normal straight
    if values == list(range(values[0], values[0] + 5)):
        return [c for c in cards if c.numeric_value == values[4]][0].rank
    
    # Ace-low straight (A-2-3-4-5)
    if values == [2, 3, 4, 5, 14]:
        return Rank.FIVE  # 5-high straight
    
    return None

# ============================================================================
# JOKER RESOLUTION
# ============================================================================

@dataclass
class JokerTransform:
    position: int
    target_card: Card

def resolve_jokers(board: List[Card]) -> Tuple[List[Card], List[JokerTransform]]:
    """
    Resolve Jokers to optimal cards for maximum payout.
    
    Returns:
        - Final board with Jokers replaced
        - List of transformations
    """
    joker_positions = [i for i, c in enumerate(board) if c.is_joker]
    
    if not joker_positions:
        return board, []
    
    if len(joker_positions) > 2:
        raise ValueError(f"Max 2 Jokers allowed, got {len(joker_positions)}")
    
    # Get available cards (not on board)
    used_cards = {str(c) for c in board if not c.is_joker}
    available = [c for c in get_all_standard_cards() if str(c) not in used_cards]
    
    best_payout = -1
    best_transforms = []
    
    if len(joker_positions) == 1:
        # Single Joker: try all 48 cards
        for card in available:
            test_board = board.copy()
            test_board[joker_positions[0]] = card
            
            hand = evaluate_hand(test_board)
            payout = hand.multiplier
            
            if payout > best_payout:
                best_payout = payout
                best_transforms = [JokerTransform(joker_positions[0], card)]
    
    else:
        # Two Jokers: try all combinations
        for i, card1 in enumerate(available):
            for card2 in available[i+1:]:
                test_board = board.copy()
                test_board[joker_positions[0]] = card1
                test_board[joker_positions[1]] = card2
                
                hand = evaluate_hand(test_board)
                payout = hand.multiplier
                
                if payout > best_payout:
                    best_payout = payout
                    best_transforms = [
                        JokerTransform(joker_positions[0], card1),
                        JokerTransform(joker_positions[1], card2),
                    ]
    
    # Apply transformations
    final_board = board.copy()
    for transform in best_transforms:
        final_board[transform.position] = transform.target_card
    
    return final_board, best_transforms

# ============================================================================
# GAME ROUND SIMULATION
# ============================================================================

def simulate_round() -> Dict[str, Any]:
    """
    Simulate single round - this is what Math SDK will call.
    
    Returns a "book" in Stake Engine format.
    """
    # Deal 5 cards
    deck = create_deck()
    random.shuffle(deck)
    initial_board = deck[:5]
    
    # Resolve Jokers
    final_board, joker_transforms = resolve_jokers(initial_board)
    
    # Evaluate hand
    hand = evaluate_hand(final_board)
    
    # Generate bookEvents for frontend
    book_events = []
    index = 0
    
    # Event 1: Reveal initial board
    book_events.append({
        "index": index,
        "type": "reveal_initial_board",
        "board": [{"symbol": str(card)} for card in initial_board]
    })
    index += 1
    
    # Event 2: Joker transform (if any)
    if joker_transforms:
        book_events.append({
            "index": index,
            "type": "joker_transform",
            "jokerTransforms": [
                {
                    "position": t.position,
                    "targetSymbol": str(t.target_card)
                }
                for t in joker_transforms
            ]
        })
        index += 1
    
    # Event 3: Hand result
    book_events.append({
        "index": index,
        "type": "hand_result",
        "handCategory": hand.hand_type.value,
        "payoutMultiplier": hand.multiplier,
        "winningPositions": hand.winning_positions,
        "jackpot": hand.multiplier >= 40
    })
    
    # Return book (Stake Engine format)
    return {
        "id": None,  # Will be assigned by Math SDK
        "payoutMultiplier": hand.multiplier,
        "events": book_events,
        
        # Metadata for analysis
        "handType": hand.hand_type.value,
        "hasJoker": len(joker_transforms) > 0,
        "initialBoard": [str(c) for c in initial_board],
        "finalBoard": [str(c) for c in final_board],
    }
