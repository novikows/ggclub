"""
Simulation runner for RTP calculation and outcome generation.

Run: python -m games.joker_poker.simulate
"""

import sys
from collections import Counter
from .game import simulate_round, GAME_CONFIG, HandType

def calculate_rtp(num_rounds: int = 1_000_000):
    """
    Calculate RTP by simulating many rounds.
    
    This validates that our paytable gives correct RTP.
    """
    print(f"\n{'='*70}")
    print(f"RTP SIMULATION - Joker Poker Board")
    print(f"{'='*70}")
    print(f"Simulating {num_rounds:,} rounds...\n")
    
    total_bet = 0
    total_win = 0
    hand_counts = Counter()
    joker_count = 0
    
    # Simulate rounds
    for i in range(num_rounds):
        if (i + 1) % 100_000 == 0:
            print(f"Progress: {i+1:,}/{num_rounds:,} ({(i+1)/num_rounds*100:.1f}%)")
        
        book = simulate_round()
        
        # Track stats
        total_bet += 1  # Normalized to 1 unit bet
        total_win += book["payoutMultiplier"]
        hand_counts[book["handType"]] += 1
        
        if book["hasJoker"]:
            joker_count += 1
    
    # Calculate RTP
    rtp = (total_win / total_bet) * 100 if total_bet > 0 else 0
    
    # Print results
    print(f"\n{'='*70}")
    print(f"RESULTS")
    print(f"{'='*70}\n")
    
    print(f"Total rounds: {num_rounds:,}")
    print(f"Total bet: {total_bet:,} units")
    print(f"Total win: {total_win:,.2f} units")
    print(f"\n🎯 RTP: {rtp:.2f}% (target: {GAME_CONFIG['target_rtp']}%)\n")
    
    # Hand frequency
    print(f"{'='*70}")
    print(f"HAND FREQUENCIES")
    print(f"{'='*70}\n")
    
    # Sort by rarity (rarest first)
    hand_order = [
        HandType.ROYAL_FLUSH.value,
        HandType.STRAIGHT_FLUSH.value,
        HandType.FOUR_OF_A_KIND.value,
        HandType.FULL_HOUSE.value,
        HandType.FLUSH.value,
        HandType.STRAIGHT.value,
        HandType.THREE_OF_A_KIND.value,
        HandType.TWO_PAIR.value,
        HandType.PAIR.value,
        HandType.HIGH_CARD.value,
    ]
    
    for hand_type_str in hand_order:
        count = hand_counts.get(hand_type_str, 0)
        freq = (count / num_rounds) * 100 if num_rounds > 0 else 0
        print(f"{hand_type_str:20} : {count:8,} ({freq:6.3f}%)")
    
    # Joker frequency
    joker_freq = (joker_count / num_rounds) * 100 if num_rounds > 0 else 0
    print(f"\n{'='*70}")
    print(f"Joker rounds: {joker_count:,} ({joker_freq:.2f}%)")
    print(f"Expected: ~3.7% (1 in 27 rounds)")
    print(f"{'='*70}\n")
    
    # Validation
    target_rtp = GAME_CONFIG['target_rtp']
    tolerance = 1.0  # ±1%
    
    if abs(rtp - target_rtp) > tolerance:
        print(f"⚠️  WARNING: RTP {rtp:.2f}% is outside target range {target_rtp}% ±{tolerance}%")
        print(f"   Consider adjusting paytable multipliers.\n")
    else:
        print(f"✅ RTP is within target range!\n")
    
    return rtp

def generate_outcomes(num_outcomes: int = 10_000_000):
    """
    Generate outcomes for upload to Stake Engine.
    
    This creates the pre-simulated outcomes that RGS will use.
    """
    print(f"\n{'='*70}")
    print(f"GENERATING OUTCOMES")
    print(f"{'='*70}")
    print(f"Generating {num_outcomes:,} outcomes...\n")
    print(f"This may take 30-60 minutes...\n")
    
    outcomes = []
    
    for i in range(num_outcomes):
        if (i + 1) % 1_000_000 == 0:
            print(f"Progress: {i+1:,}/{num_outcomes:,} ({(i+1)/num_outcomes*100:.1f}%)")
        
        book = simulate_round()
        outcomes.append(book)
    
    print(f"\n✅ Generated {len(outcomes):,} outcomes!")
    print(f"\nNext step: Upload to Stake Engine")
    print(f"Run: make upload GAME=joker_poker\n")
    
    return outcomes

def main():
    """Main entry point"""
    import argparse
    
    parser = argparse.ArgumentParser(description='Joker Poker Simulation')
    parser.add_argument('--rounds', type=int, default=1_000_000,
                       help='Number of rounds for RTP calculation')
    parser.add_argument('--generate', action='store_true',
                       help='Generate outcomes for Stake Engine')
    parser.add_argument('--count', type=int, default=10_000_000,
                       help='Number of outcomes to generate')
    
    args = parser.parse_args()
    
    if args.generate:
        # Generate outcomes
        generate_outcomes(args.count)
    else:
        # Calculate RTP
        calculate_rtp(args.rounds)

if __name__ == "__main__":
    main()
